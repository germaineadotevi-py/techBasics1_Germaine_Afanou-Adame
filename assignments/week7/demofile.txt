import sys
import time
import json
from datetime import datetime

# --- Assignment 7 Configuration ---
DEBUG = True  # Toggle this to False to play the actual game

LEADERBOARD_FILE = "leaderboard.json"

# Global state
inventory = []
items_in_room = [
    {"name": "bread", "description": "A fresh piece of bread. It looks very nourishing and gives you vital energy."},
    {"name": "key", "description": "A heavy iron key. It looks like it fits the exit door perfectly."}
]

has_eaten_bread = False  # Stores whether the player has boosted their health
start_time = 0.0

# --- Functions ---

def show_inventory():
    if not inventory:
        print("Your inventory is empty.")
    else:
        print("--- Your Inventory ---")
        for item in inventory:
            print(f"- {item['name']}")

def show_room_items():
    if not items_in_room:
        print("The room is completely empty.")
    else:
        print("--- Items in the Room ---")
        for item in items_in_room:
            print(f"- {item['name']}")

def pick_up(item_name):
    for item in items_in_room:
        if item["name"] == item_name:
            inventory.append(item)
            items_in_room.remove(item)
            print(f"You picked up: {item_name}")

            if item_name == "key":
                open_door_and_attack()
            return

    print(f"There is no '{item_name}' here.")

def use(item_name):
    global has_eaten_bread

    is_in_inventory = any(item["name"] == item_name for item in inventory)

    if not is_in_inventory:
        print(f"You don't have '{item_name}' in your inventory to use.")
        return

    if item_name == "bread":
        has_eaten_bread = True
        inventory[:] = [item for item in inventory if item["name"] != "bread"]
        print("\n[HEALTH +100]")
        print("You eat the bread. You feel extremely strong and your health is fully restored!")
    else:
        print(f"You don't know how to use '{item_name}' here.")

def examine(item_name):
    for item in inventory:
        if item["name"] == item_name:
            print(f"[Inventory] {item['name'].upper()}: {item['description']}")
            return
    for item in items_in_room:
        if item["name"] == item_name:
            print(f"[Room] {item['name'].upper()}: {item['description']}")
            return

    print(f"You don't see any '{item_name}' here.")

def open_door_and_attack():
    print("\n--------------------------------------------------")
    print("You put the key into the lock... THE DOOR OPENS!")
    print("Suddenly, an evil creature jumps out of the shadows and attacks you instantly!")
    print("--------------------------------------------------\n")

    if has_eaten_bread:
        elapsed_time = round(time.time() - start_time, 2)
        print("BAM! The attack hits you hard!")
        print("But because you ate the nourishing bread beforehand, you have enough health and survive!")
        print(f"\n🎉 CONGRATULATIONS! You managed to escape in {elapsed_time} seconds! 🎉")

        # Save records because the player won
        manage_records(elapsed_time)
    else:
        print("OH NO! The attack catches you completely off guard!")
        print("You don't have enough health points to withstand the blow...")
        print("\n💀 GAME OVER! You have been defeated. You should have eaten something first... 💀")

    sys.exit()

# --- Assignment 7: Record Saving & Leaderboard System ---

def manage_records(time_used):
    print("\nSaving your epic escape details...")
    name = input("Enter your name for the leaderboard: ").strip()
    if not name:
        name = "Anonymous"

    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    new_record = {
        "name": name,
        "timestamp": timestamp,
        "time_used": time_used
    }

    records = []

    # Try loading existing records
    try:
        with open(LEADERBOARD_FILE, "r") as file:
            records = json.load(file)
            print("[System Log] Existing record file found and loaded successfully.")
    except FileNotFoundError:
        print("[System Log] No existing data found. Creating a new record file...")
    except json.JSONDecodeError:
        print("[System Log] Error reading existing file (corrupted data). Starting fresh...")
    except Exception as e:
        print(f"[System Log] Unexpected error loading records: {e}")

    # Append new data
    records.append(new_record)

    # Sort records: lower time = faster escape = better rank
    records = sorted(records, key=lambda x: x["time_used"])

    # Try saving the updated data
    try:
        with open(LEADERBOARD_FILE, "w") as file:
            json.dump(records, file, indent=4)
    except Exception as e:
        print(f"Error saving your record to disk: {e}")
        return

    # Display Leaderboard
    display_leaderboard(records)

def display_leaderboard(records):
    print("\n====================================================")
    print("                 🏆 LEADERBOARD 🏆                  ")
    print("       (Ranked by fastest escape time)              ")
    print("====================================================")
    print(f"{'Rank':<6}{'Player Name':<20}{'Time (s)':<12}{'Timestamp':<20}")
    print("-" * 58)

    for idx, row in enumerate(records, start=1):
        print(f"{idx:<6}{row['name']:<20}{row['time_used']:<12.2f}{row['timestamp']:<20}")
    print("====================================================\n")

# --- Main Game Loop ---

def game_loop():
    global start_time

    # Assignment 7 Debug Check
    if DEBUG:
        print("!!! DEBUG MODE ACTIVE !!!")
        print("Skipping main gameplay. Running test record routine.")
        mock_score_time = 12.34  # Placeholder testing time
        manage_records(mock_score_time)
        print("Debug testing complete. Exiting.")
        sys.exit()

    # Normal Gameplay Route
    print("====================================================")
    print("Welcome to the Escape Room!")
    print("Find a way out. Quick, you are being timed!")
    print("You look around and see the following items on a table:")
    print(" - bread")
    print(" - key")
    print("====================================================")
    print("Commands: inventory, look, pickup [item], use [item], examine [item], quit")

    start_time = time.time()  # Start timing the user

    while True:
        command_input = input("\n> ").strip().lower()
        parts = command_input.split(' ', 1)
        command = parts[0]
        argument = parts[1] if len(parts) > 1 else None

        match command:
            case "help":
                print("Commands: inventory, look, pickup [item], use [item], examine [item], quit")
            case "inventory":
                show_inventory()
            case "look":
                show_room_items()
            case "pickup":
                if argument: pick_up(argument)
                else: print("What do you want to pick up? (e.g., pickup bread)")
            case "use":
                if argument: use(argument)
                else: print("What do you want to use? (e.g., use bread)")
            case "examine":
                if argument: examine(argument)
                else: print("What do you want to examine? (e.g., examine key)")
            case "quit":
                print("Thanks for playing!")
                break
            case _:
                print("Unknown command. Type 'help' to see the list of commands.")

if __name__ == "__main__":
    game_loop()
